   775→// Helper component for nutrient bar
   776→function NutrientBar({ label, value, status }: { label: string; value: number; status: string }) {
   777→  const statusColor = status === 'optimal'
   778→    ? 'bg-emerald-500'
   779→    : status === 'low'
   780→    ? 'bg-amber-500'
   781→    : 'bg-red-500';
   782→  const barWidth = Math.min(100, (value / 500) * 100);
   783→  return (
   784→    <div>
   785→      <div className='flex items-center justify-between text-sm mb-1'>
   786→        <span>{label}</span>
   787→        <span className='text-muted-foreground'>{value} kg/ha</span>
   788→      </div>
   789→      <div className='h-2 rounded-full bg-muted overflow-hidden'>
   790→        <div className={`h-full rounded-full ${statusColor} transition-all`} style={{ width: `${barWidth}%` }} />
   791→      </div>
   792→    </div>
   793→  );
   794→}
