   421→    setLoginError('');

   423→      await login(role);

   426→      setLoginError('Backend unavailable. Using offline mode.');

   431→    setLoginError('');

   433→      try {
   434→        await loginWithPhone(`+91 ${phone}`, password);
   435→        const user = useAuthStore.getState().user;
   436→        const route = user?.role === 'consumer' ? 'consumer/dashboard' : 'farmer/dashboard';

   439→        setLoginError('Invalid phone or password');

   441→    }
   442→  };
   443→
   444→  const handleEmailLogin = async () => {
   445→    setLoginError('');
   446→    if (email.includes('@') && password.length > 0) {
   447→      try {
   448→        await loginWithEmail(email, password);
   449→        const user = useAuthStore.getState().user;
   450→        const route = user?.role === 'consumer' ? 'consumer/dashboard' : 'farmer/dashboard';
   451→        navigate(route);
   452→      } catch {
   453→        setLoginError('Invalid email or password');
   454→      }
