    38→function getCategoryIcon(category: string) {
    39→  for (const [key, icon] of Object.entries(categoryIcons)) {
    40→    if (category.toLowerCase().includes(key.toLowerCase())) return icon;
    41→  }
    42→  return '📦';
    43→}

    45→function getCategoryColor(category: string) {
    46→  const map: Record<string, string> = {
    47→    Vegetables: 'bg-emerald-50 text-emerald-700 border-emerald-200',
    48→    Grains: 'bg-amber-50 text-amber-700 border-amber-200',

   461→// ============== 3. PRODUCTS PAGE ==============
   462→type SortOption = 'price_asc' | 'price_desc' | 'rating' | 'newest';
   463→const allProductCategories = ['All', 'Vegetables', 'Grains', 'Spices', 'Dairy', 'Organic', 'Fruits', 'Fibers'];
   464→
   465→export function ProductsPage() {
   466→  const { navigate } = useAppStore();
   467→  const { addToCart } = useCartStore();
   468→  const [search, setSearch] = useState('');
   469→  const [sort, setSort] = useState<SortOption>('newest');
   470→  const [category, setCategory] = useState('All');
   471→
   472→  const allProducts = useMemo(() => [...mockConsumerProducts, ...mockSeedProducts], []);
   473→
   474→  const filtered = useMemo(() => {
   475→    let products = allProducts;
   476→    if (category !== 'All') {
   477→      products = products.filter(p => p.category === category);
   478→    }
   479→    if (search) {
   480→      const q = search.toLowerCase();
   481→      products = products.filter(
   482→        p => p.name.toLowerCase().includes(q) || p.category.toLowerCase().includes(q) || (p.farmer && p.farmer.toLowerCase().includes(q))
   483→      );
   484→    }
   485→    switch (sort) {
   486→      case 'price_asc': products = [...products].sort((a, b) => a.price - b.price); break;
   487→      case 'price_desc': products = [...products].sort((a, b) => b.price - a.price); break;
   488→      case 'rating': products = [...products].sort((a, b) => b.rating - a.rating); break;
   489→      case 'newest': break;
   490→    }
   491→    return products;
   492→  }, [search, sort, category, allProducts]);
   493→
   494→  return (
   495→    <div className='space-y-6'>
   496→      <PageHeader title='All Products' subtitle='Browse fresh produce and farm supplies' />
   497→
   498→      {/* Search + Sort Row */}
   499→      <div className='flex flex-col sm:flex-row gap-3'>
   500→        <div className='relative flex-1'>
   501→          <Search className='absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground' />
   502→          <Input
   503→            placeholder='Search all products...'
   504→            className='pl-10 rounded-xl h-11'
   505→            value={search}
   506→            onChange={e => setSearch(e.target.value)}
   507→          />
   508→        </div>
   509→        <Select value={sort} onValueChange={v => setSort(v as SortOption)}>
