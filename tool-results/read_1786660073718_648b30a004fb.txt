   778→// Helper component for nutrient bar
   779→function NutrientBar({ label, value, status }: { label: string; value: number; status: string }) {
   780→  const statusColor = status === 'optimal'
   781→    ? 'bg-emerald-500'
   782→    : status === 'low'
   783→    ? 'bg-amber-500'
   784→    : 'bg-red-500';
   785→  const barWidth = Math.min(100, (value / 500) * 100);
   786→  return (
   787→    <div>
   788→      <div className='flex items-center justify-between text-sm mb-1'>
   789→        <span>{label}</span>
   790→        <span className='text-muted-foreground'>{value} kg/ha</span>
   791→      </div>
   792→      <div className='h-2 rounded-full bg-muted overflow-hidden'>
   793→        <div className={`h-full rounded-full ${statusColor} transition-all`} style={{ width: `${barWidth}%` }} />
   794→      </div>
   795→    </div>
   796→  );
   797→}
